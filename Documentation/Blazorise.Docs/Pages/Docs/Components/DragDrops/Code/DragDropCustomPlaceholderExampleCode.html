@code {
    public class DropItem
    {
        public string Name { get; init; }

        public string Group { get; set; }
    }

    private static readonly string[] DropZoneNames = new[] { "Backlog", "Done" };

    private List<DropItem> items = new()
    {
        new DropItem() { Name = "Write proposal", Group = "Backlog" },
        new DropItem() { Name = "Review design", Group = "Backlog" },
        new DropItem() { Name = "Create prototype", Group = "Done" },
    };

    private Task ItemDropped( DraggableDroppedEventArgs<DropItem> dropItem )
    {
        dropItem.Item.Group = dropItem.DropZoneName;

        return Task.CompletedTask;
    }
}
