public sealed class StatusBadgeReportElementPlugin : IReportElementPlugin, IReportElementPdfRenderer
{
    public const string TypeName = "docs.status-badge";

    public ReportElementDescriptor Descriptor { get; } = new()
    {
        TypeName = TypeName,
        DisplayName = "Status Badge",
        Category = "Custom",
        Icon = IconName.Tag,
        Width = 150,
        Height = 30,
    };

    public Type RendererComponentType => typeof( StatusBadgeReportElementRenderer );

    public Type PropertiesComponentType => typeof( StatusBadgeReportElementProperties );

    public IReportElementPdfRenderer PdfRenderer => this;

    public ReportCustomElementDefinition CreateElement()
    {
        return new()
        {
            Properties = new()
            {
                ["text"] = "Status",
            },
        };
    }

    public IEnumerable<PdfElementDefinition> Render( ReportElementPdfRenderContext context )
    {
        string text = GetText( context.Element );
        double width = Math.Min( context.Element.Width, Math.Max( 36, text.Length * 5.5 + 16 ) );
        double height = Math.Min( context.Element.Height, 16 );

        yield return new()
        {
            Type = PdfElementType.Text,
            X = Math.Max( 0, ( context.Element.Width - width ) / 2 ),
            Y = Math.Max( 0, ( context.Element.Height - height ) / 2 ),
            Width = width,
            Height = height,
            Text = text,
            Font = new()
            {
                Bold = true,
                Size = 9,
                Alignment = TextAlignment.Center,
                VerticalAlignment = VerticalAlignment.Middle,
                Color = "#FFFFFF",
            },
            Border = new()
            {
                Width = 0,
            },
            Appearance = new()
            {
                BackgroundColor = "#28A745",
            },
        };
    }

    private static string GetText( ReportCustomElementDefinition element )
        => element?.Properties?["text"]?.GetValue<string>() ?? "Status";
}
